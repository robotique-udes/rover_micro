#ifndef __LOG_H__
#define __LOG_H__

#if !defined(ARDUINO_ESP32S3_DEV)
#error CPU not supported
#endif

#if defined(VERBOSE)
#include "rover_lib2/helpers/macros.hpp"
#include <Stream.h>

namespace Logger
{
    Stream& g_loggerOutput __attribute__((weak)) = Serial;

    enum class eSeverityLevels : size_t
    {
        DEBUG_ = 0,
        INFO,
        WARN,
        ERROR,
        FATAL,
        eLAST
    };
}  // namespace Logger

#if !defined(SEVERITY_LEVEL)
#define SEVERITY_LEVEL Logger::eSeverityLevels::DEBUG_
#endif

namespace
{
    const char* COLOR_RESET = "\033[97m";
    constexpr std::array<const char*, static_cast<size_t>(Logger::eSeverityLevels::eLAST)> SEVERITY_NAMES
        = {"DEBUG", "INFO", "WARN", "ERROR", "FATAL"};
    constexpr std::array<const char*, static_cast<size_t>(Logger::eSeverityLevels::eLAST)> SEVERITY_COLORS = {
        "\033[90m",  // DEBUG -> Gray
        "\033[97m",  // INFO -> White
        "\033[33m",  // WARN -> Yellow
        "\033[31m",  // ERROR -> Red
        "\033[31m"   // FATAL -> Red
    };

    constexpr const char* getSeverityColor(Logger::eSeverityLevels severity_)
    {
        return SEVERITY_COLORS[TO_UNDERLYING(severity_)];
    }

    constexpr const char* getSeverityName(Logger::eSeverityLevels severity_)
    {
        return SEVERITY_NAMES[TO_UNDERLYING(severity_)];
    }

    constexpr bool isSeverityEnabled(Logger::eSeverityLevels severity_)
    {
        return TO_UNDERLYING(severity_) >= TO_UNDERLYING(static_cast<decltype(severity_)>(SEVERITY_LEVEL));
    }

    template<Logger::eSeverityLevels SEVERITY>
    constexpr void logInternal(const char* format, ...)
    {
        if constexpr (isSeverityEnabled(SEVERITY))
        {
            va_list args;
            va_start(args, format);

            Logger::g_loggerOutput.printf("%s[%lu][%s]%s(%d): ",
                                          getSeverityColor(SEVERITY),
                                          millis(),
                                          getSeverityName(SEVERITY),
                                          __FILENAME__,
                                          __LINE__);

            Logger::g_loggerOutput.vprintf(format, args);
            Logger::g_loggerOutput.printf("\n%s", COLOR_RESET);

            va_end(args);
        }
    }
}  // namespace

namespace LOG
{
    template<typename... Args>
    constexpr void DEBUG(const char* format, Args... args)
    {
        logInternal<Logger::eSeverityLevels::DEBUG_>(format, args...);
    }

    template<typename... Args>
    constexpr void INFO(const char* format, Args... args)
    {
        logInternal<Logger::eSeverityLevels::INFO>(format, args...);
    }

    template<typename... Args>
    constexpr void WARN(const char* format, Args... args)
    {
        logInternal<Logger::eSeverityLevels::WARN>(format, args...);
    }

    template<typename... Args>
    constexpr void ERROR(const char* format, Args... args)
    {
        logInternal<Logger::eSeverityLevels::ERROR>(format, args...);
    }

    template<typename... Args>
    constexpr void FATAL(const char* format, Args... args)
    {
        logInternal<Logger::eSeverityLevels::FATAL>(format, args...);
    }
}  // namespace LOG

#else  // defined(VERBOSE)
namespace LOG

    }

    }

    {
    }

    constexpr void FATAL(const char* format, Args... args)
    {
    }
}  // namespace LOG
